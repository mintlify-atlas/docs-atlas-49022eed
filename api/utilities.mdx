Closure Next provides a comprehensive set of utility functions for common operations including type checking, DOM manipulation, object handling, and string transformation.

## Type Checking Utilities

### isString()

Checks if a value is a string.

```typescript
function isString(value: unknown): value is string
```

<ParamField path="value" type="unknown" required>
  The value to check
</ParamField>

**Returns:** `boolean` - Type guard that returns true if value is a string

**Example:**
```typescript
import { isString } from '@closure/core';

const value: unknown = 'hello';
if (isString(value)) {
  console.log(value.toUpperCase()); // TypeScript knows it's a string
}
```

## DOM Utilities

### addClass()

Adds a CSS class to an element.

```typescript
function addClass(element: HTMLElement, className: string): void
```

<ParamField path="element" type="HTMLElement" required>
  The element to add the class to
</ParamField>

<ParamField path="className" type="string" required>
  The CSS class name to add
</ParamField>

**Example:**
```typescript
import { addClass } from '@closure/core';

const button = document.querySelector('button');
if (button) {
  addClass(button, 'active');
}
```

<ParamField path="element" type="HTMLElement" required>

<ParamField path="className" type="string" required>

### hasClass()

Checks if an element has a CSS class.

```typescript
function hasClass(element: HTMLElement, className: string): boolean
```

<ParamField path="element" type="HTMLElement" required>
  The element to check
</ParamField>

<ParamField path="className" type="string" required>
  The CSS class name to check for
</ParamField>

**Returns:** `boolean` - True if the element has the class

**Example:**
```typescript
import { hasClass } from '@closure/core';

const button = document.querySelector('button');
if (button && hasClass(button, 'active')) {
  console.log('Button is active');
}
```

### toggleClass()

<ParamField path="className" type="string" required>

## Object Utilities

### isEmpty()

Checks if a value is empty (null, undefined, empty array, empty object, or empty string).

```typescript
function isEmpty(value: unknown): boolean
```

<ParamField path="value" type="unknown" required>
  The value to check
</ParamField>

**Returns:** `boolean` - True if the value is considered empty

**Example:**
```typescript
import { isEmpty } from '@closure/core';

console.log(isEmpty(null)); // true
console.log(isEmpty(undefined)); // true
console.log(isEmpty([])); // true
console.log(isEmpty({})); // true
console.log(isEmpty('')); // true
console.log(isEmpty([1])); // false
console.log(isEmpty({ key: 'value' })); // false
```

### deepClone()

Creates a deep copy of an object or array.

```typescript
function deepClone<T>(obj: T): T
```

<ParamField path="obj" type="T" required>
  The object or array to clone
</ParamField>

<ParamField path="T" type="type parameter">
  The type of the object being cloned
</ParamField>

**Returns:** `T` - A deep copy of the input

**Example:**
```typescript
import { deepClone } from '@closure/core';

const original = {
  name: 'John',
  address: {
    city: 'New York',
    zip: '10001'
  },
  tags: ['user', 'admin']
};

const copy = deepClone(original);
copy.address.city = 'Boston';
copy.tags.push('verified');

console.log(original.address.city); // 'New York' - unchanged
console.log(original.tags); // ['user', 'admin'] - unchanged
console.log(copy.address.city); // 'Boston'
console.log(copy.tags); // ['user', 'admin', 'verified']
```

## String Utilities

### camelToKebab()

Converts a camelCase string to kebab-case.

```typescript
function camelToKebab(str: string): string
```

<ParamField path="str" type="string" required>
  The camelCase string to convert
</ParamField>

**Returns:** `string` - The kebab-case version of the string

**Example:**
```typescript
import { camelToKebab } from '@closure/core';

console.log(camelToKebab('backgroundColor')); // 'background-color'
console.log(camelToKebab('fontSize')); // 'font-size'
console.log(camelToKebab('myComponentName')); // 'my-component-name'
```

## Combined Usage Examples

### Type-Safe Data Processing

```typescript
import { isArray, isObject, isString, deepClone } from '@closure/core';

function processData(data: unknown) {
  if (isArray(data)) {
    // TypeScript knows data is an array
    return data.map(item => processData(item));
  }
  
  if (isObject(data)) {
    // TypeScript knows data is an object
    return deepClone(data);
  }
  
  if (isString(data)) {
    // TypeScript knows data is a string
    return data.toUpperCase();
  }
  
  return data;
}
```

### Dynamic Class Management

```typescript
import { hasClass, addClass, removeClass } from '@closure/core';

function setActiveState(element: HTMLElement, active: boolean) {
  const hasActive = hasClass(element, 'active');
  
  if (active && !hasActive) {
    addClass(element, 'active');
  } else if (!active && hasActive) {
    removeClass(element, 'active');
  }
}
```

### CSS Property Transformation

```typescript
import { camelToKebab, kebabToCamel } from '@closure/core';

function applyStyles(element: HTMLElement, styles: Record<string, string>) {
  Object.entries(styles).forEach(([key, value]) => {
    // Convert camelCase to kebab-case for CSS
    const cssProperty = camelToKebab(key);
    element.style.setProperty(cssProperty, value);
  });
}

applyStyles(document.body, {
  backgroundColor: '#fff',
  fontSize: '16px',
  lineHeight: '1.5'
});
```
