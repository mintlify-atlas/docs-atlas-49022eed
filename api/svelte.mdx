## Overview

The `@closure-next/svelte` package provides Svelte-specific integration for Closure Next components using Svelte stores and lifecycle hooks.

## Installation

```bash
npm install @closure-next/svelte
```

## createClosureStore

```typescript
function createClosureStore<T extends Component>(
  ComponentClass: new (domHelper?: DOMHelper) => T,
  options?: ClosureComponentOptions<T>
): ClosureStore<T>
```

Creates a Svelte store that manages the lifecycle of a Closure Next component, integrating with Svelte's reactivity system.

### Parameters

<ParamField path="ComponentClass" type="ComponentConstructor<T>" required>
  The Closure Next component class constructor.
</ParamField>

<ParamField path="options" type="ClosureComponentOptions<T>">
  Configuration options for the component wrapper.
  
  **Properties**:
  - `props`: ComponentState (optional) - Initial props for the component
  - `ssrOptions`: Object (optional) - SSR configuration with hydration strategy
  - `onMount`: (component: T) => void (optional) - Called after component mounts
  - `onDestroy`: (component: T) => void (optional) - Called before component destroys
  - `onError`: (error: Error) => void (optional) - Error handler
</ParamField>

### Returns

<ResponseField name="store" type="ClosureStore<T>">
  A Svelte store object containing reactive state and control methods.

  **Properties**:
  - `component`: Writable\<T | null\> - Writable store containing the component instance
  - `props`: Writable\<ComponentState\> - Writable store for component props
  - `element`: Writable\<HTMLElement | null\> - Writable store for the DOM element
  - `setProps`: (props: Partial\<ComponentState\>) => Promise\<void\> - Method to update props
</ResponseField>

### Example

```svelte
<script lang="ts">
  import { createClosureStore } from '@closure-next/svelte';
  import { Component, DOMHelper } from '@closure-next/core';

  class Counter extends Component {
    private count = 0;

    protected async createDom(): Promise<void> {
      this.element = this.domHelper.createElement('div');
      this.updateDisplay();
    }

    private updateDisplay() {
      if (this.element) {
        this.element.textContent = `Count: ${this.count}`;
      }
    }

    increment() {
      this.count++;
      this.updateDisplay();
    }
  }

  const store = createClosureStore(Counter, {
    props: {},
    onMount: (component) => {
      console.log('Component mounted', component);
    }
  });

  const { component, element } = store;

  function handleClick() {
    const comp = $component;
    if (comp && 'increment' in comp) {
      (comp as Counter).increment();
    }
  }
</script>

<div bind:this={$element}>
  <!-- Closure component renders here -->
</div>
<button on:click={handleClick}>Increment</button>
```

## ClosureComponentOptions

```typescript
interface ClosureComponentOptions<T extends Component = Component> {
  props?: ComponentState;
  ssrOptions?: {
    hydration?: 'client-only' | 'server-first' | 'progressive';
    ssr?: boolean;
  };
  onMount?: (component: T) => void;
  onDestroy?: (component: T) => void;
  onError?: (error: Error) => void;
}
```

Configuration options for creating a Closure component in Svelte.

### Properties

<ParamField path="props" type="ComponentState">
  Initial props to pass to the component.
</ParamField>

<ParamField path="ssrOptions" type="object">

</ParamField>

<ParamField path="onMount" type="(component: T) => void">
  Callback invoked after the component mounts.
</ParamField>

<ParamField path="onDestroy" type="(component: T) => void">

</ParamField>

<ParamField path="onError" type="(error: Error) => void">

## ClosureStore

```typescript
interface ClosureStore<T extends Component = Component> {
  component: Writable<T | null>;
  props: Writable<ComponentState>;
  element: Writable<HTMLElement | null>;
  setProps(props: Partial<ComponentState>): Promise<void>;
}
```

Svelte store interface returned by `createClosureStore`.

### Store Properties

<ParamField path="component" type="Writable<T | null>">
  Writable store containing the component instance. Subscribe to track component state.
</ParamField>

<ParamField path="props" type="Writable<ComponentState>">
  Writable store for component props. Changes automatically propagate to the component.
</ParamField>

<ParamField path="element" type="Writable<HTMLElement | null>">
  Writable store for the DOM element where the component is rendered.
</ParamField>

<ParamField path="setProps" type="(props: Partial<ComponentState>) => Promise<void>">
  Async method to update component props. Calls setter methods on the component instance.
</ParamField>

### Example with Reactive Props

```svelte
<script lang="ts">
  import { createClosureStore } from '@closure-next/svelte';
  import { writable } from 'svelte/store';
  import { MyComponent } from './my-component';

  const store = createClosureStore(MyComponent);
  const { component, props, element, setProps } = store;

  // Reactive prop updates
  const title = writable('Initial Title');
  
  $: setProps({ title: $title });
</script>

<input bind:value={$title} />
<div bind:this={$element}></div>
```

## Server-Side Rendering

### SSR with SvelteKit

```svelte
<!-- +page.svelte -->
<script lang="ts">
  import { createClosureStore } from '@closure-next/svelte';
  import { browser } from '$app/environment';
  import { MyComponent } from './my-component';

  const store = createClosureStore(MyComponent, {
    ssrOptions: {
      ssr: !browser,
      hydration: 'server-first'
    }
  });

  const { element } = store;
</script>

<div bind:this={$element}></div>
```

### Hydration Strategies

<Accordion title="client-only">
  Renders only on the client. No server-side rendering. Best for components that require browser APIs.
</Accordion>

<Accordion title="server-first">
  Renders on the server first, then hydrates on the client. Best for SEO-critical content.
</Accordion>

<Accordion title="progressive">
  Partially hydrates the component, prioritizing interactive elements. Best for large components.
</Accordion>

## Advanced Patterns

### Component with Lifecycle Hooks

```svelte
<script lang="ts">
  import { createClosureStore } from '@closure-next/svelte';
  import { onMount as svelteOnMount, onDestroy as svelteOnDestroy } from 'svelte';

  const store = createClosureStore(MyComponent, {
    onMount: (component) => {
      console.log('Closure component mounted');
      // Initialize external resources
    },
    onDestroy: (component) => {
      console.log('Closure component destroying');
      // Cleanup external resources
    },
    onError: (error) => {
      console.error('Component error:', error);
      // Handle errors
    }
  });
</script>
```

### Integration with Svelte Stores

```svelte
<script lang="ts">
  import { createClosureStore } from '@closure-next/svelte';
  import { derived } from 'svelte/store';

  const store = createClosureStore(MyComponent);
  const { component, props } = store;

  // Derive computed values
  const isReady = derived(component, $comp => $comp !== null);

  // Sync with external store
  import { globalState } from './stores';
  
  $: if ($globalState.theme) {
    setProps({ theme: $globalState.theme });
  }
</script>

{#if $isReady}
  <div bind:this={$element}></div>
{:else}
  <p>Loading...</p>
{/if}
```
