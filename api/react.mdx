## Overview

The `@closure-next/react` package provides a seamless integration between Closure Next components and React applications through component wrappers and SSR support.

## Installation

```bash
npm install @closure-next/react
```

## ClosureComponent

```typescript
function ClosureComponent<T extends Component>(props: {
  component: T;
  domHelper: DOMHelper;
  children?: React.ReactNode;
}): JSX.Element
```

A React component that renders and manages the lifecycle of a Closure Next component.

### Props

<ParamField path="component" type="T extends Component" required>
  The Closure Next component instance to render.
</ParamField>

<ParamField path="domHelper" type="DOMHelper" required>
  The DOMHelper instance for DOM operations.
</ParamField>

<ParamField path="children" type="React.ReactNode">
  Optional React children to render alongside the Closure component.
</ParamField>

### Behavior

- **Rendering**: Renders the Closure component into the DOM when mounted
- **Lifecycle**: Automatically disposes the component when unmounted
- **Children**: Renders React children alongside the Closure component

### Example

```tsx
import { Component, DOMHelper } from '@closure-next/react';
import { ClosureComponent } from '@closure-next/react';

class MyClosureComponent extends Component {
  protected async createDom(): Promise<void> {
    this.element = this.domHelper.createElement('div');
    this.element.textContent = 'Hello from Closure Next!';
  }
}

function MyReactComponent() {
  const component = new MyClosureComponent(new DOMHelper());
  
  return (
    <ClosureComponent 
      component={component} 
      domHelper={new DOMHelper()}
    />
  );
}
```

### With State Management

```tsx
import React, { useState } from 'react';
import { Component, DOMHelper } from '@closure-next/react';
import { ClosureComponent } from '@closure-next/react';

class Counter extends Component {
  protected async createDom(): Promise<void> {
    this.element = this.domHelper.createElement('div');
    this.updateDisplay();
  }

  private updateDisplay(): void {
    if (this.element) {
      const count = this.getState().count || 0;
      this.element.textContent = `Count: ${count}`;
    }
  }

  async setState(state: any): Promise<void> {
    await super.setState(state);
    this.updateDisplay();
  }
}

function CounterApp() {
  const [counter] = useState(() => new Counter(new DOMHelper()));
  
  const increment = () => {
    const count = counter.getState().count || 0;
    counter.setState({ count: count + 1 });
  };

  return (
    <div>
      <ClosureComponent 
        component={counter} 
        domHelper={new DOMHelper()}
      />
      <button onClick={increment}>Increment</button>
    </div>
  );
}
```

## ServerComponent

```typescript
class ServerComponent extends Component {
  async renderToString(): Promise<string>
  async hydrate(container?: HTMLElement): Promise<void>
  enableSSR(): void
  disableSSR(): void
  isSSREnabled(): boolean
}
```

A component class designed for server-side rendering in React applications.

### Methods

<ParamField path="renderToString" type="() => Promise<string>">
  Renders the component to an HTML string for server-side rendering.
</ParamField>

<ParamField path="hydrate" type="(container?: HTMLElement) => Promise<void>">
  Hydrates a server-rendered component on the client side.
</ParamField>

<ParamField path="enableSSR" type="() => void">
  Enables SSR for this component instance.
</ParamField>

<ParamField path="disableSSR" type="() => void">
  Disables SSR for this component instance.
</ParamField>

<ParamField path="isSSREnabled" type="() => boolean">
  Returns whether SSR is enabled for this component.
</ParamField>

### Server-Side Example

```tsx
import { ServerComponent, DOMHelper } from '@closure-next/react';
import { JSDOM } from 'jsdom';

// Server-side rendering
const dom = new JSDOM('<!DOCTYPE html><html><body></body></html>');
const domHelper = new DOMHelper(dom.window.document);

class MySSRComponent extends ServerComponent {
  protected async createDom(): Promise<void> {
    this.element = this.domHelper.createElement('div');
    this.element.textContent = 'SSR Content';
  }
}

const component = new MySSRComponent(domHelper);
component.enableSSR();

const html = await component.renderToString();
// Returns: <div>SSR Content</div>
```

### Client-Side Hydration

```tsx
import { ServerComponent, DOMHelper } from '@closure-next/react';

class MySSRComponent extends ServerComponent {
  // ... component implementation
}

// Client-side
const domHelper = new DOMHelper();
const component = new MySSRComponent(domHelper);
await component.hydrate(document.getElementById('ssr-root'));
```

## Best Practices

### Component Lifecycle

- Always create a new `DOMHelper` instance for each component
- Dispose components when unmounting to prevent memory leaks
- Use `ServerComponent` for SSR scenarios

### TypeScript Integration

```tsx
import type { ComponentState } from '@closure-next/react';

interface MyComponentState extends ComponentState {
  title: string;
  count: number;
}

class MyTypedComponent extends Component {
  async setState(state: MyComponentState): Promise<void> {
    await super.setState(state);
  }

  getState(): MyComponentState {
    return super.getState() as MyComponentState;
  }
}
```

### Error Handling

```tsx
import { ClosureComponent, Component, DOMHelper } from '@closure-next/react';
import { ErrorBoundary } from 'react-error-boundary';

function App() {
  return (
    <ErrorBoundary fallback={<div>Error loading component</div>}>
      <ClosureComponent 
        component={new MyComponent(new DOMHelper())} 
        domHelper={new DOMHelper()}
      />
    </ErrorBoundary>
  );
}
```
