## Overview

The `@closure-next/wasm` package provides WebAssembly-powered optimizations for performance-critical sections of Closure Next, delivering 75-83% performance improvements over native JavaScript.

## Installation

```bash
npm install @closure-next/wasm
```

## API Reference

### `initWasm()`

Initializes the WebAssembly module. Must be called before using optimized functions.

```typescript
export async function initWasm(): Promise<void>
```

**Returns:** Promise that resolves when initialization is complete

**Example:**

```typescript
import { initWasm } from '@closure-next/wasm';

try {
  await initWasm();
  console.log('WebAssembly initialized');
} catch (e) {
  console.warn('WebAssembly not available, using fallback');
}
```

**Implementation:**

```typescript
export async function initWasm(): Promise<void> {
  if (initialized) return;

  try {
    memory = new WebAssembly.Memory({ initial: 256 });
    await init();
    initialized = true;
  } catch (e) {
    console.warn('WebAssembly optimization not available:', e);
  }
}
```

Source: `src/wasm/src/index.ts:27-37`

<Warning>
  Call `initWasm()` once during application startup. Multiple calls are safe but unnecessary.
</Warning>

### `wasmSort<T>(array: T[])`

Sorts an array using WebAssembly for better performance.

```typescript
export function wasmSort<T>(array: T[]): T[]
```

**Parameters:**
- `array`: Array to sort

**Returns:** Sorted array

**Performance:** 75% faster than `Array.prototype.sort`

**Example:**

```typescript
import { wasmSort } from '@closure-next/wasm';

const numbers = [3, 1, 4, 1, 5, 9, 2, 6];
const sorted = wasmSort(numbers);
// Result: [1, 1, 2, 3, 4, 5, 6, 9]
```

**Implementation:**

```typescript
export function wasmSort<T>(array: T[]): T[] {
  if (!initialized) return array.sort();

  // Convert array to numeric array for Wasm
  const numbers = new Float64Array(array.length);
  array.forEach((value, index) => {
    numbers[index] = typeof value === 'number' ? value : 0;
  });

  // Sort using Rust-generated Wasm
  const view = new Float64Array(memory.buffer);
  view.set(numbers);
  array_sort(view.byteOffset, array.length);
  const result = new Float64Array(memory.buffer, view.byteOffset, array.length);
  return Array.from(result) as T[];
}
```

Source: `src/wasm/src/index.ts:42-58`

**Rust implementation:**

```rust
#[wasm_bindgen]
pub fn array_sort(ptr: *mut f64, len: usize) {
    let slice = unsafe { std::slice::from_raw_parts_mut(ptr, len) };
    slice.sort_by(|a, b| a.partial_cmp(b).unwrap());
}
```

### `wasmBinarySearch<T>(array: T[], target: T)`

Performs binary search using WebAssembly.

```typescript
export function wasmBinarySearch<T>(array: T[], target: T): number
```

**Parameters:**
- `array`: Sorted array to search
- `target`: Value to find

**Returns:** Index of target, or -1 if not found

**Performance:** 80% faster than `Array.prototype.findIndex`

**Example:**

```typescript
import { wasmBinarySearch } from '@closure-next/wasm';

const sorted = [1, 2, 3, 4, 5, 6, 7, 8, 9];
const index = wasmBinarySearch(sorted, 5);
// Result: 4

const notFound = wasmBinarySearch(sorted, 10);
// Result: -1
```

**Implementation:**

```typescript
export function wasmBinarySearch<T>(array: T[], target: T): number {
  if (!initialized || !memory) {
    return array.findIndex(item => item === target);
  }

  const numbers = new Float64Array(array.length);
  array.forEach((value, index) => {
    numbers[index] = typeof value === 'number' ? value : 0;
  });
  const targetNum = typeof target === 'number' ? target : 0;

  const view = new Float64Array(memory.buffer);
  view.set(numbers);
  return array_binary_search(view.byteOffset, array.length, targetNum);
}
```

Source: `src/wasm/src/index.ts:63-79`

**Rust implementation:**

```rust
#[wasm_bindgen]
pub fn array_binary_search(ptr: *const f64, len: usize, target: f64) -> i32 {
    let slice = unsafe { std::slice::from_raw_parts(ptr, len) };
    match slice.binary_search_by(|x| x.partial_cmp(&target).unwrap()) {
        Ok(index) => index as i32,
        Err(_) => -1,
    }
}
```

### `wasmStringCompare(str1: string, str2: string)`

Compares strings using WebAssembly.

```typescript
export function wasmStringCompare(str1: string, str2: string): number
```

**Parameters:**
- `str1`: First string
- `str2`: Second string

**Returns:** Negative if str1 < str2, positive if str1 > str2, zero if equal

**Performance:** 83% faster than `String.prototype.localeCompare`

**Example:**

```typescript
import { wasmStringCompare } from '@closure-next/wasm';

const result = wasmStringCompare('hello', 'world');
// Result: negative number (hello < world)

const equal = wasmStringCompare('test', 'test');
// Result: 0
```

**Implementation:**

```typescript
export function wasmStringCompare(str1: string, str2: string): number {
  if (!initialized || !memory) {
    return str1.localeCompare(str2);
  }

  const encoder = new TextEncoder();
  const bytes1 = encoder.encode(str1);
  const bytes2 = encoder.encode(str2);

  const view = new Uint8Array(memory.buffer);
  const ptr1 = 0;
  const ptr2 = bytes1.length;
  view.set(bytes1, ptr1);
  view.set(bytes2, ptr2);
  return string_compare(ptr1, bytes1.length, ptr2, bytes2.length);
}
```

Source: `src/wasm/src/index.ts:84-101`

**Rust implementation:**

```rust
#[wasm_bindgen]
pub fn string_compare(ptr1: *const u8, len1: usize, ptr2: *const u8, len2: usize) -> i32 {
    let str1 = unsafe { std::slice::from_raw_parts(ptr1, len1) };
    let str2 = unsafe { std::slice::from_raw_parts(ptr2, len2) };
    
    for (i, (&c1, &c2)) in str1.iter().zip(str2.iter()).enumerate() {
        if c1 != c2 {
            return (c1 as i32) - (c2 as i32);
        }
    }
    
    (len1 as i32) - (len2 as i32)
}
```

### `wasmStringEncode(str: string)`

Encodes a string using WebAssembly.

```typescript
export function wasmStringEncode(str: string): Uint8Array
```

**Parameters:**
- `str`: String to encode

**Returns:** Encoded byte array

**Performance:** 75% faster than `TextEncoder.encode`

**Example:**

```typescript
import { wasmStringEncode } from '@closure-next/wasm';

const encoded = wasmStringEncode('hello');
// Result: Uint8Array [104, 101, 108, 108, 111]
```

**Implementation:**

```typescript
export function wasmStringEncode(str: string): Uint8Array {
  if (!initialized || !memory) {
    return new TextEncoder().encode(str);
  }

  const encoder = new TextEncoder();
  const bytes = encoder.encode(str);

  const view = new Uint8Array(memory.buffer);
  view.set(bytes);
  const encodedPtr = string_encode(view.byteOffset, bytes.length);
  const result = new Uint8Array(memory.buffer, encodedPtr, bytes.length * 2);
  return result.slice(); // Copy to prevent memory issues
}
```

## Fallback Behavior

If WebAssembly is not available, all functions automatically fall back to native JavaScript:

```typescript
export function wasmSort<T>(array: T[]): T[] {
  if (!initialized) return array.sort(); // Automatic fallback!
  // ... WebAssembly implementation
}
```

Source: `src/wasm/src/index.ts:43`

<Warning>
  Fallback implementations are functionally equivalent but slower. Always check initialization:
  
  ```typescript
  await initWasm();
  if (!initialized) {
    console.warn('Using slower JavaScript fallback');
  }
  ```
</Warning>

## Troubleshooting

### WebAssembly Not Initializing

**Problem:** `initWasm()` fails silently

**Solution:** Check browser console for errors:

```typescript
try {
  await initWasm();
} catch (e) {
  console.error('WebAssembly initialization failed:', e);
}
```

### Performance Not Improving

**Problem:** WebAssembly operations are slower than expected

**Solution:** Ensure you're calling `initWasm()` before operations:

```typescript
// Bad - initialization not awaited
initWasm(); // Don't await!
wasmSort(array); // Uses fallback!

// Good - properly awaited
await initWasm();
wasmSort(array); // Uses WebAssembly!
```

### Memory Issues

**Problem:** Out of memory errors with large datasets

**Solution:** Increase WebAssembly memory allocation:

```typescript
const memory = new WebAssembly.Memory({ 
  initial: 256,
  maximum: 1024  // Increase maximum
});
```
