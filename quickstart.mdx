## Overview

This guide walks you through creating a simple interactive component with state management. You'll learn the core concepts of Closure Next: components, DOM helpers, and event handling.

## Building Your First Component

<Steps>

### Import Core Classes

Start by importing the necessary classes from `@closure-next/core`:

```typescript app.ts
import { Component, DOMHelper, EventType } from '@closure-next/core';
import type { ComponentState } from '@closure-next/core';
```

### Create a Custom Component

Extend the `Component` base class to create a counter component:

```typescript app.ts
class CounterComponent extends Component {
  protected count: number;

  constructor(domHelper: DOMHelper, initialCount: number = 0) {
    super(domHelper);
    this.count = initialCount;
    this.state = { count: initialCount };
  }

  protected async createDom(): Promise<void> {
    await super.createDom();
    
    if (this.element) {
      this.element.innerHTML = `
        <div style="padding: 20px; font-family: sans-serif;">
          <h2>Counter: ${this.state.count}</h2>
          <button id="increment">Increment</button>
          <button id="decrement">Decrement</button>
        </div>
      `;
      
      // Attach event listeners
      this.element.querySelector('#increment')?.addEventListener('click', () => {
        this.increment();
      });
      
      this.element.querySelector('#decrement')?.addEventListener('click', () => {
        this.decrement();
      });
    }
  }

  async increment(): Promise<void> {
    this.count++;
    await this.setState({ count: this.count });
  }

  async decrement(): Promise<void> {
    this.count--;
    await this.setState({ count: this.count });
  }

  async setState(state: ComponentState): Promise<void> {
    if ('count' in state) {
      this.count = state.count;
    }
    await super.setState(state);
  }
}
```

### Initialize and Render

Create an instance and render it to the DOM:

```typescript app.ts
// Initialize the DOM helper
const domHelper = new DOMHelper();

// Create the component
const counter = new CounterComponent(domHelper, 0);

// Set an ID for easy access
counter.setId('my-counter');

// Listen to state changes
counter.addEventListener(EventType.STATECHANGE, (event) => {
  console.log('State changed:', event.data);
});

// Render to the page
const container = document.getElementById('app');
if (container) {
  counter.render(container);
}
```

## Understanding the Code

### Component Lifecycle

Closure Next components follow a clear lifecycle:

1. **Constructor** - Initialize state and dependencies
2. **`createDom()`** - Build the DOM structure
3. **`render()`** - Mount to a container element
4. **`setState()`** - Update state and trigger re-render
5. **`dispose()`** - Clean up when done

### State Management

The `setState()` method is asynchronous and triggers automatic re-rendering:

```typescript
await component.setState({ count: 42 });
```

<Note>
State changes emit a `STATECHANGE` event automatically. Subscribe to this event to react to updates.
</Note>

### Type Safety

All core APIs are fully typed:

```typescript
import type { 
  ComponentInterface,
  ComponentState,
  EventInterface,
  EventTargetInterface 
} from '@closure-next/core';
```

## Parent-Child Components

Components can be composed into hierarchies:

```typescript
const parent = new Component(domHelper);
const child = new Component(domHelper);

parent.addChild(child);

// Render parent, children render automatically
await parent.render(container);

console.log(child.getParent()); // parent
console.log(parent.getChildren()); // [child]
```

<Tip>
When you dispose a parent component, all children are automatically disposed as well.
</Tip>

## Server-Side Rendering

Closure Next supports SSR out of the box:

```typescript
import { JSDOM } from 'jsdom';

// Create a server-side DOM
const jsdom = new JSDOM('<!DOCTYPE html><html><body></body></html>');
const document = jsdom.window.document;

// Create component with server-side DOM
const domHelper = new DOMHelper(document);
const component = new CounterComponent(domHelper);

// Render to string
const html = await component.renderToString();

// Send HTML to client
res.send(html);
```

On the client, hydrate the component:

```typescript
const domHelper = new DOMHelper();
const component = new CounterComponent(domHelper);

// Hydrate existing server-rendered HTML
await component.hydrate(container);
```

## Complete Example

Here's the full working example:

```typescript app.ts
import { Component, DOMHelper, EventType } from '@closure-next/core';
import type { ComponentState } from '@closure-next/core';

class CounterComponent extends Component {
  protected count: number;

  constructor(domHelper: DOMHelper, initialCount: number = 0) {
    super(domHelper);
    this.count = initialCount;
    this.state = { count: initialCount };
  }

  protected async createDom(): Promise<void> {
    await super.createDom();
    
    if (this.element) {
      this.element.innerHTML = `
        <div style="padding: 20px; font-family: sans-serif;">
          <h2>Counter: ${this.state.count}</h2>
          <button id="increment">Increment</button>
          <button id="decrement">Decrement</button>
        </div>
      `;
      
      this.element.querySelector('#increment')?.addEventListener('click', () => {
        this.increment();
      });
      
      this.element.querySelector('#decrement')?.addEventListener('click', () => {
        this.decrement();
      });
    }
  }

  async increment(): Promise<void> {
    this.count++;
    await this.setState({ count: this.count });
  }

  async decrement(): Promise<void> {

// Initialize
const domHelper = new DOMHelper();

```
